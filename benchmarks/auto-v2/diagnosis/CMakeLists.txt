cmake_minimum_required(VERSION 3.24)
project(MasketJSONRelocationDiagnosis LANGUAGES CXX)
add_executable(TokenRelocation TokenRelocation.cpp ../../BenchmarkUtils.cpp)
target_compile_features(TokenRelocation PRIVATE cxx_std_17)
target_include_directories(TokenRelocation PRIVATE ../../../parser ../..)
target_compile_definitions(TokenRelocation PRIVATE DIAG_CONFIGURATION="$<CONFIG>")
if(MSVC)
    target_compile_options(TokenRelocation PRIVATE /W4)
else()
    target_compile_options(TokenRelocation PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
add_test(NAME Diagnosis.RelocationArithmetic COMMAND TokenRelocation --self-test)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/build-info-$<CONFIG>.txt"
    CONTENT "compiler=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\nconfiguration=$<CONFIG>\nflags=${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_RELEASE}\n")
