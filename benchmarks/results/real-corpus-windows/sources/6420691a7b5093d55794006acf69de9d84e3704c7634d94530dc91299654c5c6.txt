find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(AUTO_V3_SPEED_BUDGET 4194304 CACHE STRING "Speculative speed reserve bytes (geometric floor may override)")
set(AUTO_V3_MEMORY_BUDGET 1048576 CACHE STRING "Speculative memory reserve bytes (geometric floor may override)")
foreach(value AUTO_V3_SPEED_BUDGET AUTO_V3_MEMORY_BUDGET)
    if(NOT "${${value}}" MATCHES "^[0-9]+$" OR ${value} GREATER 1073741824)
        message(FATAL_ERROR "${value} must be an integer in 0..1073741824")
    endif()
endforeach()
set(AUTO_V3_SPEED_QUARTERS 8 CACHE STRING "Pressure growth floor in quarters")
set(AUTO_V3_MEMORY_QUARTERS 8 CACHE STRING "Pressure growth floor in quarters")
foreach(value AUTO_V3_SPEED_QUARTERS AUTO_V3_MEMORY_QUARTERS)
    if(NOT "${${value}}" MATCHES "^[678]$")
        message(FATAL_ERROR "${value} must be 6, 7 or 8")
    endif()
endforeach()
set(AUTO_V3_SPEED_LOOKAHEAD 0 CACHE STRING "Include the next inevitable relocation in pressure")
set(AUTO_V3_MEMORY_LOOKAHEAD 0 CACHE STRING "Include the next inevitable relocation in pressure")
foreach(value AUTO_V3_SPEED_LOOKAHEAD AUTO_V3_MEMORY_LOOKAHEAD)
    if(NOT "${${value}}" MATCHES "^[01]$")
        message(FATAL_ERROR "${value} must be 0 or 1")
    endif()
endforeach()
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generate_lexer.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_lexer.py"
    "${PROJECT_SOURCE_DIR}/parser/JsonLexer.cpp" "${generated}" COMMAND_ERROR_IS_FATAL ANY)
set(v3_programs)
foreach(goal speed memory)
    foreach(mode time trace profile)
        if(mode STREQUAL "profile" AND NOT TARGET ParserProfile)
            continue()
        endif()
        set(library AutoV3-${goal}-${mode})
        add_library(${library} STATIC "${generated}/${goal}.cpp" $<TARGET_OBJECTS:AutoParser>)
        target_link_libraries(${library} PUBLIC MasketJSON)
        target_include_directories(${library} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/benchmarks/auto" "${PROJECT_SOURCE_DIR}/benchmarks/reservation")
        target_compile_definitions(${library} PUBLIC JsonLexer=AutoLexer JsonParser=AutoParser
            AUTO_V3_SPEED_BYTES=${AUTO_V3_SPEED_BUDGET} AUTO_V3_MEMORY_BYTES=${AUTO_V3_MEMORY_BUDGET}
            AUTO_V3_SPEED_QUARTERS=${AUTO_V3_SPEED_QUARTERS} AUTO_V3_MEMORY_QUARTERS=${AUTO_V3_MEMORY_QUARTERS}
            AUTO_V3_SPEED_LOOKAHEAD=${AUTO_V3_SPEED_LOOKAHEAD} AUTO_V3_MEMORY_LOOKAHEAD=${AUTO_V3_MEMORY_LOOKAHEAD})
        if(mode STREQUAL "time")
            add_executable(AutoV3Time-${goal} ../reservation/Timing.cpp ../BenchmarkUtils.cpp)
            target_compile_definitions(AutoV3Time-${goal} PRIVATE MATRIX_CONFIGURATION="$<CONFIG>")
            target_link_libraries(AutoV3Time-${goal} PRIVATE ${library})
            add_executable(AutoV3Inspect-${goal} ../reservation/Inspect.cpp)
            target_link_libraries(AutoV3Inspect-${goal} PRIVATE ${library})
            add_executable(AutoV3Verify-${goal} Verify.cpp ../auto/Compare.cpp $<TARGET_OBJECTS:AutoBaseline>)
            target_include_directories(AutoV3Verify-${goal} PRIVATE "${PROJECT_SOURCE_DIR}/fuzz")
            target_link_libraries(AutoV3Verify-${goal} PRIVATE ${library})
            add_executable(AutoV3Replay-${goal} "${PROJECT_SOURCE_DIR}/fuzz/FuzzReplay.cpp" ../auto/Compare.cpp $<TARGET_OBJECTS:AutoBaseline>)
            target_include_directories(AutoV3Replay-${goal} PRIVATE "${PROJECT_SOURCE_DIR}/fuzz")
            target_link_libraries(AutoV3Replay-${goal} PRIVATE ${library})
            list(APPEND v3_programs AutoV3Time-${goal} AutoV3Inspect-${goal} AutoV3Verify-${goal} AutoV3Replay-${goal})
        elseif(mode STREQUAL "trace")
            target_compile_definitions(${library} PUBLIC MASKETJSON_AUTO_V3_TRACE)
            add_executable(AutoV3Trace-${goal} Trace.cpp)
            target_link_libraries(AutoV3Trace-${goal} PRIVATE ${library})
            list(APPEND v3_programs AutoV3Trace-${goal})
        else()
            target_compile_definitions(${library} PUBLIC MASKETJSON_RESERVATION_STATS)
            add_executable(AutoV3Profile-${goal} ../ParserProfile.cpp ../MemoryTracker.cpp)
            target_link_libraries(AutoV3Profile-${goal} PRIVATE ${library} Threads::Threads)
            list(APPEND v3_programs AutoV3Profile-${goal})
        endif()
    endforeach()
endforeach()
add_custom_target(AutoV3Tools DEPENDS ${v3_programs} AutoTime-balanced AutoTime-low-memory AutoInspect-balanced AutoInspect-low-memory)
if(TARGET ParserProfile)
    add_dependencies(AutoV3Tools AutoProfile-balanced AutoProfile-low-memory)
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<IF:$<BOOL:${CMAKE_CONFIGURATION_TYPES}>,$<CONFIG>/,>build-info-$<CONFIG>.txt"
    CONTENT "compiler=${CMAKE_CXX_COMPILER_ID}\ncompiler_version=${CMAKE_CXX_COMPILER_VERSION}\nconfiguration=$<CONFIG>\nspeed_budget_bytes=${AUTO_V3_SPEED_BUDGET}\nmemory_budget_bytes=${AUTO_V3_MEMORY_BUDGET}\nspeed_quarters=${AUTO_V3_SPEED_QUARTERS}\nmemory_quarters=${AUTO_V3_MEMORY_QUARTERS}\nspeed_lookahead=${AUTO_V3_SPEED_LOOKAHEAD}\nmemory_lookahead=${AUTO_V3_MEMORY_LOOKAHEAD}\nconfig_flags=$<$<CONFIG:Release>:${CMAKE_CXX_FLAGS_RELEASE}>$<$<CONFIG:RelWithDebInfo>:${CMAKE_CXX_FLAGS_RELWITHDEBINFO}>\n")
if(MASKETJSON_BUILD_FUZZERS)
    foreach(goal speed memory)
        add_library(AutoV3-${goal}-fuzz STATIC "${generated}/${goal}.cpp" $<TARGET_OBJECTS:AutoParserFuzz>)
        target_link_libraries(AutoV3-${goal}-fuzz PUBLIC MasketJSON)
        target_include_directories(AutoV3-${goal}-fuzz PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/benchmarks/auto")
        target_compile_definitions(AutoV3-${goal}-fuzz PUBLIC JsonLexer=AutoLexer JsonParser=AutoParser
            AUTO_V3_SPEED_BYTES=${AUTO_V3_SPEED_BUDGET} AUTO_V3_MEMORY_BYTES=${AUTO_V3_MEMORY_BUDGET}
            AUTO_V3_SPEED_QUARTERS=${AUTO_V3_SPEED_QUARTERS} AUTO_V3_MEMORY_QUARTERS=${AUTO_V3_MEMORY_QUARTERS}
            AUTO_V3_SPEED_LOOKAHEAD=${AUTO_V3_SPEED_LOOKAHEAD} AUTO_V3_MEMORY_LOOKAHEAD=${AUTO_V3_MEMORY_LOOKAHEAD})
        target_compile_options(AutoV3-${goal}-fuzz PRIVATE -fsanitize=fuzzer-no-link)
        add_executable(AutoV3Fuzz-${goal} "${PROJECT_SOURCE_DIR}/fuzz/ParserFuzz.cpp" ../auto/Compare.cpp $<TARGET_OBJECTS:AutoBaseline>)
        target_include_directories(AutoV3Fuzz-${goal} PRIVATE "${PROJECT_SOURCE_DIR}/fuzz")
        target_compile_options(AutoV3Fuzz-${goal} PRIVATE -fsanitize=fuzzer-no-link)
        target_link_options(AutoV3Fuzz-${goal} PRIVATE -fsanitize=fuzzer)
        target_link_libraries(AutoV3Fuzz-${goal} PRIVATE AutoV3-${goal}-fuzz)
    endforeach()
endif()
